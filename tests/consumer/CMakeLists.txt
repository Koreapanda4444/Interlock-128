cmake_minimum_required(VERSION 3.20)
project(InterlockExternalConsumer LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
find_package(Interlock128 0.1 CONFIG REQUIRED)
add_executable(consumer consumer.c)
target_link_libraries(consumer PRIVATE Interlock128::interlock)
enable_testing()
add_test(NAME installed_consumer COMMAND consumer)
