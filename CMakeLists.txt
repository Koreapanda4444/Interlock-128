cmake_minimum_required(VERSION 3.20)
project(Interlock128 VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
include(CTest)
include(CheckCSourceCompiles)
find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
check_c_source_compiles("#include <limits.h>\n#include <stdint.h>\n_Static_assert(CHAR_BIT == 8, \"8-bit bytes required\");\n_Static_assert(sizeof(uint32_t) == 4 && sizeof(uint64_t) == 8, \"fixed widths required\");\n#ifndef UINTPTR_MAX\n#error uintptr_t required\n#endif\nint main(void) { return 0; }" INTERLOCK_SUPPORTED_INTEGERS)
if(NOT INTERLOCK_SUPPORTED_INTEGERS)
  message(FATAL_ERROR "Interlock requires 8-bit bytes, uint32_t, uint64_t and uintptr_t")
endif()
option(CRYPTO_BUILD_RESEARCH "Build private reduced-round research tools" OFF)
option(CRYPTO_BUILD_BENCHMARKS "Build native measurements" OFF)
option(CRYPTO_ENABLE_SANITIZERS "Enable ASan and UBSan" OFF)
add_library(interlock STATIC src/support.c)
target_include_directories(interlock PRIVATE src)
target_link_libraries(interlock PRIVATE OpenSSL::Crypto)
if(MSVC)
  target_compile_options(interlock PRIVATE /W4)
else()
  target_compile_options(interlock PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
if(BUILD_TESTING)
  add_library(interlock_testcore STATIC src/interlock.c)
  target_compile_definitions(interlock_testcore PRIVATE INTERLOCK_TESTING)
  target_include_directories(interlock_testcore PRIVATE src)
  add_executable(test_platform tests/test_platform.c)
  add_test(NAME platform COMMAND test_platform)
  add_executable(test_support tests/test_support.c)
  target_include_directories(test_support PRIVATE src)
  target_link_libraries(test_support PRIVATE interlock)
  add_test(NAME support COMMAND test_support)
  add_executable(test_shake tests/test_shake.c src/support.c)
  target_compile_definitions(test_shake PRIVATE INTERLOCK_TESTING)
  target_include_directories(test_shake PRIVATE src)
  target_link_libraries(test_shake PRIVATE OpenSSL::Crypto)
  add_test(NAME shake COMMAND test_shake)
  add_executable(test_substitution tests/test_substitution.c)
  target_compile_definitions(test_substitution PRIVATE INTERLOCK_TESTING)
  target_include_directories(test_substitution PRIVATE src)
  target_link_libraries(test_substitution PRIVATE interlock_testcore)
  add_test(NAME substitution COMMAND test_substitution)
  add_executable(test_mixing tests/test_mixing.c)
  target_compile_definitions(test_mixing PRIVATE INTERLOCK_TESTING)
  target_include_directories(test_mixing PRIVATE src)
  target_link_libraries(test_mixing PRIVATE interlock_testcore)
  add_test(NAME mixing COMMAND test_mixing)
  add_executable(test_forward tests/test_forward.c)
  target_compile_definitions(test_forward PRIVATE INTERLOCK_TESTING)
  target_include_directories(test_forward PRIVATE src)
  target_link_libraries(test_forward PRIVATE interlock_testcore)
  add_test(NAME forward COMMAND test_forward)
endif()
